// src/user-service.ts (or wherever you handle user data)

import { onAuthStateChanged } from 'firebase/auth';
import { ref, update } from 'firebase/database';
import { auth, database } from './firebase-config'; // Import the initialized services

/**
 * Adds or updates the nickname for the currently authenticated user in Realtime Database.
 * If no user is signed in, it logs a message.
 * @param nickname The nickname string to set for the user.
 */
export async function setCurrentUserNickname(nickname: string): Promise<void> {
  // onAuthStateChanged is the most reliable way to get the current user,
  // as it ensures the authentication state has been fully loaded.
  onAuthStateChanged(auth, async (user) => {
    if (user) {
      // User is signed in, so we can get their unique ID (UID)
      const userId = user.uid;

      // Create a reference to the specific user's data path
      // This will be "users/<user_uid>"
      const userRef = ref(database, `users/${userId}`);

      try {
        // Use `update()` to set the nickname.
        // `update()` is great because it only modifies the specified fields
        // and leaves other fields at `userRef` (like email, if you had them) untouched.
        // If 'nickname' doesn't exist, it creates it.
        await update(userRef, {
          nickname: nickname,
        });

        console.log(`Successfully set nickname "${nickname}" for user: ${userId}`);
      } catch (error: any) {
        // Handle any errors during the database write operation
        console.error("Error setting nickname:", error.message);
        // You might want to show a user-friendly error message here
      }
    } else {
      // No user is signed in
      console.warn("Cannot set nickname: No user is currently signed in.");
      // You might want to redirect to a sign-in page or show a message to the user
    }
  });
}
